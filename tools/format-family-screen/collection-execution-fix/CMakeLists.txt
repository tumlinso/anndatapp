cmake_minimum_required(VERSION 3.28)
project(collection_execution_fix LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp 1.0 CONFIG REQUIRED)
add_executable(execution-time ../collection-matrix/cpp_probe.cpp)
add_executable(execution-count ../collection-overhead/cpp_probe.cpp)
target_link_libraries(execution-time PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
target_link_libraries(execution-count PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
