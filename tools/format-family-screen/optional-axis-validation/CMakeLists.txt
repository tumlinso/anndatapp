cmake_minimum_required(VERSION 3.28)
project(optional_axis_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp 1.0 CONFIG REQUIRED)
option(HAS_AXIS_NAME_OPTION "Build with the optional axis name checking API" OFF)
add_executable(optional-axis-probe cpp_probe.cpp)
target_compile_definitions(optional-axis-probe PRIVATE HAS_AXIS_NAME_OPTION=$<BOOL:${HAS_AXIS_NAME_OPTION}>)
target_link_libraries(optional-axis-probe PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
