cmake_minimum_required(VERSION 3.28)
project(format_family_screen LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp 1.0 CONFIG REQUIRED)
add_executable(format-family-screen screen.cpp)
target_link_libraries(format-family-screen PRIVATE anndatapp::anndatapp)
option(DATASET_PREPARED "Use the candidate prepared AnnDataSet API" OFF)
add_executable(dataset-screen dataset_screen.cpp)
target_compile_definitions(dataset-screen PRIVATE DATASET_PREPARED=$<BOOL:${DATASET_PREPARED}>)
target_link_libraries(dataset-screen PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
add_executable(family-screen family_screen.cpp)
target_link_libraries(family-screen PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
