cmake_minimum_required(VERSION 3.28)
project(core_screen LANGUAGES C CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT BENCHMARK_ROOT)
  message(FATAL_ERROR "Set BENCHMARK_ROOT to the repository benchmarks directory")
endif()
add_executable(core_head_to_head "${BENCHMARK_ROOT}/head_to_head/cpp_driver.cpp")
target_compile_features(core_head_to_head PRIVATE cxx_std_20)
target_compile_options(core_head_to_head PRIVATE -ffp-contract=off)
target_link_libraries(core_head_to_head PRIVATE anndatapp::anndatapp OpenSSL::Crypto)
add_executable(core_v1 "${BENCHMARK_ROOT}/v1/main.cpp" "${BENCHMARK_ROOT}/v1/core_cases.cpp" "${BENCHMARK_ROOT}/v1/storage_cases.cpp")
target_compile_features(core_v1 PRIVATE cxx_std_20)
target_compile_definitions(core_v1 PRIVATE V1_NEW_CORE=1)
target_link_libraries(core_v1 PRIVATE anndatapp::anndatapp Threads::Threads ${CMAKE_DL_LIBS})
add_library(core_v1_io SHARED "${BENCHMARK_ROOT}/v1/io_counters.c")
target_link_libraries(core_v1_io PRIVATE ${CMAKE_DL_LIBS})
add_library(core_v1_alloc SHARED "${BENCHMARK_ROOT}/v1/alloc_counters.cpp")
target_compile_features(core_v1_alloc PRIVATE cxx_std_20)
add_library(core_head_io SHARED "${BENCHMARK_ROOT}/head_to_head/io_counters.c")
target_link_libraries(core_head_io PRIVATE ${CMAKE_DL_LIBS})
if(NOT CORE_SCREEN_SOURCE)
  message(FATAL_ERROR "Set CORE_SCREEN_SOURCE to tools/format-family-screen")
endif()
add_executable(core_mmap "${CORE_SCREEN_SOURCE}/core_mmap.cpp")
target_compile_features(core_mmap PRIVATE cxx_std_20)
target_link_libraries(core_mmap PRIVATE anndatapp::anndatapp OpenSSL::Crypto)
