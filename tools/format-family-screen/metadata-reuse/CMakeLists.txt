cmake_minimum_required(VERSION 3.28)
project(metadata_collection_comparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(anndatapp 1.0 CONFIG REQUIRED)
add_executable(small-collection-cpp small_collection_cpp.cpp)
target_link_libraries(small-collection-cpp PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
add_executable(collection-control-cpp control_cpp.cpp)
target_compile_definitions(collection-control-cpp PRIVATE HAS_AXIS_NAME_OPTION=1)
target_link_libraries(collection-control-cpp PRIVATE anndatapp::anndatapp ${CMAKE_DL_LIBS})
